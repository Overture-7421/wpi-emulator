cmake_minimum_required(VERSION 3.25)
project(wpilib_root_fixture LANGUAGES CXX)
include("${WPIEMU_FIXTURE_REPO}/cmake/WpiEmuVersions.cmake")
wpiemu_pinned_version(_wpilib_version wpilib)

function(wpiemu_target_defaults target)
endfunction()

add_subdirectory(discovery)
if(WPIEMU_FIXTURE_LAYOUT STREQUAL "maven")
    set(_expected_root "${CMAKE_BINARY_DIR}/_wpilib/${_wpilib_version}")
else()
    set(_expected_root "${WPIEMU_FIXTURE_ROOT}")
endif()
if(NOT WPIEMU_WPILIB_ROOT STREQUAL _expected_root)
    message(FATAL_ERROR "Resolved root not propagated: '${WPIEMU_WPILIB_ROOT}' != '${_expected_root}'")
endif()
get_property(_cached_root CACHE WPIEMU_WPILIB_ROOT PROPERTY VALUE)
if(NOT _cached_root STREQUAL WPIEMU_FIXTURE_CACHE_ROOT)
    message(FATAL_ERROR "User cache input changed: '${_cached_root}' != '${WPIEMU_FIXTURE_CACHE_ROOT}'")
endif()
if(NOT TARGET wpiemu_wpilib_hal OR NOT TARGET wpiemu_wpilib_wpiutil)
    message(FATAL_ERROR "Imported WPILib targets missing")
endif()

add_subdirectory("${WPIEMU_FIXTURE_REPO}/sdk" sdk)
add_subdirectory("${WPIEMU_FIXTURE_REPO}/bridges" bridges)
if(NOT TARGET wpiemu_bridges_hal_halsim)
    message(FATAL_ERROR "Real HAL bridge binding missing")
endif()
get_target_property(_definitions wpiemu_bridges_vendor INTERFACE_COMPILE_DEFINITIONS)
if(NOT "WPIEMU_VENDOR_HAS_WPILIB=1" IN_LIST _definitions)
    message(FATAL_ERROR "Vendor HAL backend missing")
endif()
get_target_property(_builtin_definitions wpiemu_bridges_builtin COMPILE_DEFINITIONS)
if(NOT "WPIEMU_BUILTIN_HAL_HALSIM" IN_LIST _builtin_definitions)
    message(FATAL_ERROR "Aggregator did not select the real HAL binding")
endif()
message(STATUS "WPILib root fixture: bindings enabled and cache preserved (tests=${WPIEMU_BUILD_TESTS})")
